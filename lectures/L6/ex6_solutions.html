Recall that the Kaplan-Meier Estimate is given by $$\hat{S}_j = \prod_{k \leq j}{(\frac{n - k}{n - k + 1})}^{d_{(k)}}$$ where $n$ is the toal number of subjects, $j$ is the event , and $d_{(k)}$ is an indicates if the event is censored or not.

def kaplan_meier(data: pd.DataFrame) -> pd.DataFrame:
    # Assume data is sorted by 'Time'
    
    # Initialize variables
    n = data.shape[0] - 1
    survival_prob = 1
    survival_probs = np.ones(n + 1)
    survival_probs[0] = survival_prob
    
    # Calculate survival probabilities
    for i in range(n):
        if data.iloc[i + 1]['Event'] == 1:  # If death occured, include data in estimation. If censored, skip
            survival_prob *= (n - i - 1) / (n - i) # Looks slightly different from formula because of 0-indexing
        survival_probs[i + 1] = survival_prob
    
    data['Survival P'] = survival_probs
    return data



data = kaplan_meier(pd.DataFrame(data))

Recall that the Log Rank Test models each time step as a Hypergeometric distribution. Hypergeometric distributions describe the number of successes $k$ in $n$ trials with a population size of $N$ without replacement.

In our case $n$ = the number of individuals in group A, $k$ = the number of deaths in group A, and $N$ = all remaining participants.

Thus we estimate mean and variance from the Hypergeometric distribution. $$\mu = \frac{n_A n_d}{n}$$ $$\sigma^{2} = \frac{n_A n_B n_d n_s}{n^2(n - 1)}$$

def log_rank_test(data: pd.DataFrame) -> ([float], [float]):

    # Sort data by time
    data = data.sort_values(by='Time').reset_index(drop=True)

    # Initialize variables
    n = data.shape[0]
    _mu = []
    _var = []

    # Hardcoded :(
    n_A = 3
    n_B = 3
    n_total = n_A + n_B

    for i in range(n):
        group = data.iloc[i]['Group']

        if data.iloc[i]['Event'] == 1: # If event is 1, then 1 person died, and n_total - 1 people hae survived

            # Calculate expectation and death of hypergeometric distribution
            _mu.append(n_A * 1 / n_total)
            _var.append(n_A * n_B * 1 * (n_total - 1) / (n_total**2 * (n_total - 1)))

            # Update parameters
            n_total -= 1
            if group == 'A':
                n_A -= 1
            else:
                n_B -= 1
            
        elif data.iloc[i]['Event'] == 0:
            if group == 'A':
                n_A -= 1
            else:
                n_B -= 1

            n_total -= 1

    return _mu, _var


def z_score(y_i: [float], mu_i: [float], var_i: [float]) -> float:
    z = np.sum((np.array(y_i) - np.array(mu_i))) / np.sum(np.sqrt(var_i))
    return z

Problem 3¶

a) Write the hazard function for the treated and control groups.
b) What is the hazard ratio between the treated and control group.
c) Interpret the hazard ratio in plain langauge.
d) If the baseline hazard at time t is $h_0(t) = 0.01$ , what is the hazard for a treated subject at time $t$?

a) From the Cox Proportional Hazard model, $$h_i(t) = h_0(t)e^{c_i \beta}$$ In our case, $c_i$ is a 1-dimensional vector that is either 0 or 1.

thus for the control group: $h_{\text{control}}(t) = h_0(t)$

and for the treatment group: $h_{\text{treatment}}(t) = h_0(t)e^{-.693}$

b) The hazard ratio is the quotient of two hazard rates, $\frac{h_{\text{treatment}}}{h_{\text{control}}}$. Our hazard rates are defined above. When taking the quotient, the baseline hazard rates disappears and we are left with $$e^{-.693} = .50007$$

c) This suggests that at any given time, the probability that an individual in the treatment group dies is half of their control counterpart.

d) If the baseline hazard rate is .01, then notice that the hazard function no longer depends on $t$.
For a treated individual, the hazard rate at time t is $.01 * e^{-.693}$ = $.05$

# Parameters for truncation
truncation_limit = 50000

# Initial guesses for mean and variance
mu_est = 40000  # Initial guess for mean
sigma_est = 10000  # Initial guess for standard deviation

# To track convergence
mean_estimates = [mu_est]
variance_estimates = [sigma_est**2]

# EM Algorithm parameters
tolerance = 10  # Convergence tolerance
max_iter = 100  # Maximum number of iterations

# EM Algorithm
for iteration in range(max_iter):
    # E-step: Calculate the expected value for censored (truncated) incomes
    censored_mask = df['Income'] == truncation_limit
    uncensored_mask = df['Income'] < truncation_limit
    censored_incomes = df.loc[censored_mask, 'Income']

    # Calculate expected values for truncated observations
    if censored_mask.sum() > 0:
        z_censored = (truncation_limit - mu_est) / sigma_est
        expected_truncated = mu_est + sigma_est * norm.pdf(z_censored) / (1 - norm.cdf(z_censored))
    else:
        expected_truncated = 0

    # Combine observed and expected values
    complete_income = df['Income'].copy()
    complete_income[censored_mask] = expected_truncated

    # M-step: Recalculate mean and variance using both observed and expected incomes
    new_mu_est = complete_income.mean()
    new_sigma_est = np.sqrt(((complete_income - new_mu_est)**2).mean())

    # Store the estimates
    mean_estimates.append(new_mu_est)
    variance_estimates.append(new_sigma_est**2)

    # Check for convergence
    if abs(new_mu_est - mu_est) < tolerance and abs(new_sigma_est - sigma_est) < tolerance:
        break

    # Update parameters for the next iteration
    mu_est = new_mu_est
    sigma_est = new_sigma_est

# Plotting the convergence of the mean and variance estimates
fig, ax = plt.subplots(2, 1, figsize=(8, 6))

# Plot for mean
ax[0].plot(mean_estimates, ".")
ax[0].set_xlabel('Iteration')
ax[0].set_ylabel('Mean')
ax[0].set_title("Mean Estimate")

# Plot for variance
ax[1].plot(variance_estimates, ".", color = "orange")
ax[1].set_xlabel('Iteration')
ax[1].set_ylabel('Variance')
ax[1].set_title("Variance Estimate")

plt.tight_layout()
plt.show()
